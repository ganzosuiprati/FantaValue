CALCOLO PREZZO MASSIMO ADATTIVO // ========================================== function calcolaPrezzoMax(pma, fantamedia) { // Proporziona la quotazione iniziale al budget attuale const basePma = Number(pma) || 1; const fm = Number(fantamedia) || 6.0; let moltiplicatore = (totalBudget / 500); let stima = Math.round(basePma * 2.2 * moltiplicatore); // Bonu/Malus su Fantamedia if (fm >= 8.0) stima += Math.round(15 * moltiplicatore); else if (fm >= 7.5) stima += Math.round(8 * moltiplicatore); return Math.max(1, stima); } // ========================================== // 4. ${initial}
`; return ` ${avatarHTML} ${player.nome.toUpperCase()} ${player.squadra} ${player.ruolo} ${Number(player.fantamedia).toFixed(1)} ${player.titolarita}% ${prezzoMax} CR `; }).join(''); } // ========================================== // 5. Ottimo colpo di stabilità!`; accentColor = "#ffcc00"; // Neon Gold } // Mostra Popup Modale mostraModalValutazione(player, prezzoPagato, prezzoConsigliato, esitoTitle, esitoDesc, accentColor); // Aggiungi alla rosa myTeam.push({ ...player, prezzoPagato }); remainingBudget -= prezzoPagato; // Aggiorna Interfaccia updateUI(); } // ========================================== // 6. MODAL POPUP SYNTHWAVE / CYBERPUNK // ========================================== function mostraModalValutazione(player, pagato, consigliato, titolo, desc, colore) { // Rimuovi eventuali modali precedenti const existing = document.getElementById('fanta-eval-modal'); if (existing) existing.remove(); const modalHTML = `

${titolo}

${player.nome.toUpperCase()} (${player.squadra})

PAGATO ${pagato} CR
CONSIGLIATO ${consigliato} CR

${desc}

`; document.body.insertAdjacentHTML('beforeend', modalHTML); } // ========================================== // 7. ${item.prezzoPagato} CR
`).join(''); } } // Ricarica la tabella escludendo chi è già stato preso renderPlayersTable(); } // ========================================== // 8.
